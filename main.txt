from hub import motion_sensor, port, button, light_matrix, light, sound

motor_pair.pair(motor_pair.PAIR_1, port.F,port.B) # Pairs motors

MIN_TURN = 100

WHEEL = 15.5 #cm (wheel circumference)

def difference(angle_difference):
    """
    Calculates shortest way to wanted angle.
    Returns adjusted angle difference in the range of -180 to 180 degrees.
    """
    while angle_difference > 180:
        angle_difference -= 360
    while angle_difference < -180:
        angle_difference += 360
    return angle_difference

def yaw():
    """
    Retrieve the robot's yaw angle in app-like format.
    """
    return round(motion_sensor.tilt_angles()[0]*-0.1)

def relative():
    """
    Calculate the average traveled distance using both motors.
    Returns the average of the absolute relative positions of motors on ports B and D.
    """
    relative = int(((abs(motor.relative_position(port.B)))+(abs(motor.relative_position(port.F))))/2)
    return relative

async def turn(angle, maxspeed = 1110):
    """
    Proportional-Integral-Derivative (PID) controlled turn function.

    Turns the robot to a specific yaw angle using a PID controller.
    Includes dynamic acceleration based on error and inertia correction.
    """
    Kp, Ki, Kd = 4.5, 0.001, 1.2# Fixed PID constants
    I, last = 0, 0
    delay = 5

    while abs(yaw() - angle) > 1:
        curyaw = yaw()
        P = difference(angle - curyaw)# calculates difference between desired and current angle for proportional correction
        I += P * delay
        D = (P - last) / delay

        if P * last < 0:# reset integral if error changes sign (to prevent windup)
            I = 0

        speed = max(MIN_TURN, min(maxspeed, round(abs(P * Kp) + I * Ki + D * Kd)))# PID-controlled turning speed

        if P > 0:# if turning right
            motor_pair.move_tank(motor_pair.PAIR_1, speed, -round(speed * 0.888))# inner wheel slowed down for precision (EXPERIMENTAL NOT TESTED nor ADJUSTED)
        else:# if turning left
            motor_pair.move_tank(motor_pair.PAIR_1, -round(speed * 0.888), speed)

        last = P
        time.sleep_ms(delay)

    motor_pair.stop(motor_pair.PAIR_1, stop = motor.SMART_BRAKE)
    time.sleep_ms(75)

async def drive(cm, angle, maxspeed = 1110, minspeed = 200):
    """
    Proportional-Integral-Derivative (PID) controlled drive function.
    Max speed prevents the motors from becoming inaccurate
    Min speed prevents the robot from stalling.
    """
    motor.reset_relative_position(port.B, 0)# Resets motors to measure distance
    motor.reset_relative_position(port.F, 0)

    reverse = cm < 0
    if abs(cm) < 15:
        maxspeed = 850# Limited speed for better accuracy at short distances

    # Calculate the distance in degrees
    distance = int(abs(cm) / WHEEL * 360) - round(maxspeed / 65)

    Kp, Ki, Kd = 4.5, 0.001, 1.2# PID constants
    I, last = 0, 0

    while abs(relative()) < distance:# Drive until distance is reached
        curyaw = yaw()
        error = difference(angle - curyaw)# Difference between target and current yaw angle
        I += error
        D = error - last
        last = error

        # PID correction
        correction = round((error * Kp + I * Ki + D * Kd) / 3)
        correction = max(-80, min(correction, 80))# Limit correction range

        steering = -correction if reverse else correction# Inverted correction if moving in reverse

        # Calculate proportional speed based on remaining distance
        togo = distance - abs(relative())
        speed = min(max(int(maxspeed * togo / distance * 2), minspeed), maxspeed)

        motor_pair.move(motor_pair.PAIR_1, steering, velocity=speed if not reverse else -speed, acceleration=900)

    motor_pair.stop(motor_pair.PAIR_1, stop=motor.SMART_BRAKE)
    time.sleep_ms(30)

async def gear(gear):
    """Switches the integrated gearbox to the desired gear"""
    motor.run_to_absolute_position(port.D, (gear*90) - 49, 1050) #Moves the motor to the specific orientation, so the LEGO gear switcher part engages the right gear
    await motor.run_to_absolute_position(port.D, (gear*90) - 49, 1050)

def is_pressed(): # returns True, when the right or left button is pressed
    return button.pressed(button.LEFT) or button.pressed(button.RIGHT)

async def main():
    #resets the yaw angle for start
    motion_sensor.reset_yaw(0)
    time.sleep_ms(250)
    light_matrix.show_image(2)
    last_color = color.UNKNOWN
    color_runs = { # sensor color to run names table
        color.BLUE: bluerun,
        color.RED: redrun,
        color.GREEN: greenrun,
        color.YELLOW: yellowrun,
        color.AZURE: azurerun,
        color.MAGENTA: magentarun,
        color.BLACK: blackrun,
        color.WHITE: whiterun
    }
    while True:
        current_color = color_sensor.color(port.C)
        # check for color change
        if current_color != last_color:
            last_color = current_color
            if current_color == color.UNKNOWN:
                light.color(light.POWER, color.BLACK)
                light.color(light.CONNECT, color.BLACK)
            else:
                # update lights to match the color sensor color
                light.color(light.POWER, current_color)
                light.color(light.CONNECT, current_color)

        # start run with button press
        if is_pressed() and current_color in color_runs:
            sound.beep(440, 300) # beep to show that run has been activated
            await color_runs[current_color]()# trigger the corresponding run

        # no attachments on robot
        if current_color == color.UNKNOWN:
            if abs(difference(0 - motor.absolute_position(port.A))) >= 2 or abs(difference(0 - motor.absolute_position(port.D))) >= 2: # if no attachments are on the robot, gearbox resets to 'neutral' gear
                motor.run_to_absolute_position(port.D, 131, 1050)
                await motor.run_to_absolute_position(port.D, 131, 1050)
            if is_pressed():
                motion_sensor.reset_yaw(0)
                light_matrix.write(str(hub.temperature()))
                sound.beep(550, 300)
                await motor.run_for_degrees(port.D, 180, 1100)
                await gear(1)
                await motor.run_for_degrees(port.A, 180, 1110)
                await gear(2)
                await motor.run_for_degrees(port.A, 180, 1110)
                await gear(3)
                await motor.run_for_degrees(port.A, 180, 1110)
                await gear(4)
                await motor.run_for_degrees(port.A, 180, 1110)
runloop.run(main())
